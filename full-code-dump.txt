# React + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

import { motion } from "framer-motion";
import clsx from "clsx";
import { Link } from "react-router-dom";
import { DURATION, EASE } from "../lib/motion";

const base =
  "inline-flex items-center justify-center gap-2 px-7 py-3.5 text-[0.95rem] font-medium tracking-[0.01em] transition-colors duration-300 focus-visible:outline-2 focus-visible:outline-offset-4";

const variants = {
  primary: "bg-navy text-paper hover:bg-premier-green",
  secondary: "border border-navy text-navy hover:bg-navy hover:text-paper",
  light: "bg-paper text-navy hover:bg-premier-cyan hover:text-paper",
  ghost: "text-navy underline-offset-[6px] decoration-1",
};

/**
 * Button â€” renders a router Link when `to` is provided, otherwise a <button>.
 * variant: "primary" | "secondary" | "light" | "ghost"
 */
export default function Button({
  as,
  to,
  href,
  variant = "primary",
  className,
  children,
  ...rest
}) {
  const classes = clsx(base, variants[variant], className);

  const content =
    variant === "ghost" ? (
      <span className="relative">
        {children}
        <span className="absolute -bottom-0.5 left-0 h-px w-full origin-left scale-x-100 bg-current transition-transform duration-300 group-hover:scale-x-0" />
      </span>
    ) : (
      children
    );

  const motionProps = {
    whileHover: variant === "ghost" ? undefined : { y: -2 },
    whileTap: { scale: 0.98 },
    transition: { duration: DURATION.fast, ease: EASE },
  };

  if (to) {
    return (
      <motion.div {...motionProps} className="inline-block group">
        <Link to={to} className={classes}>
          {content}
        </Link>
      </motion.div>
    );
  }

  if (href) {
    return (
      <motion.div {...motionProps} className="inline-block group">
        <a href={href} className={classes} target="_blank" rel="noreferrer">
          {content}
        </a>
      </motion.div>
    );
  }

  const Tag = as ?? "button";
  return (
    <motion.div {...motionProps} className="inline-block group">
      <Tag className={classes} {...rest}>
        {content}
      </Tag>
    </motion.div>
  );
}
